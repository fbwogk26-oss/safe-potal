function makeLabelSprite(text, fontSize){
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  ctx.font = `700 ${fontSize}px 'Malgun Gothic', sans-serif`;
  const padX = 16;
  const w = Math.ceil(ctx.measureText(text).width) + padX*2;
  const h = fontSize + 20;
  canvas.width = w*2; canvas.height = h*2;
  ctx.scale(2,2);
  ctx.font = `700 ${fontSize}px 'Malgun Gothic', sans-serif`;
  ctx.textBaseline = 'middle';
  ctx.textAlign = 'center';
  ctx.fillStyle = 'rgba(12,15,20,0.72)';
  const r = 8;
  ctx.beginPath();
  ctx.moveTo(r,0); ctx.arcTo(w,0,w,h,r); ctx.arcTo(w,h,0,h,r); ctx.arcTo(0,h,0,0,r); ctx.arcTo(0,0,w,0,r);
  ctx.closePath(); ctx.fill();
  ctx.fillStyle = '#f2f5f8';
  ctx.fillText(text, w/2, h/2+1);
  const tex = new THREE.CanvasTexture(canvas);
  tex.minFilter = THREE.LinearFilter;
  const mat = new THREE.SpriteMaterial({map:tex, depthTest:true, depthWrite:false, transparent:true});
  const sprite = new THREE.Sprite(mat);
  sprite.scale.set(w*0.36, h*0.36, 1);
  sprite.userData.baseFontH = h*0.28;
  return sprite;
}

function initPanel(mountId, regions, opts){
  const mount = document.getElementById(mountId);
  const W = () => mount.clientWidth;
  const H = () => mount.clientHeight;

  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0x0d1117);
  scene.fog = new THREE.Fog(0x0d1117, opts.fogNear, opts.fogFar);

  const camera = new THREE.PerspectiveCamera(42, W()/H(), 1, 5000);
  const renderer = new THREE.WebGLRenderer({antialias:true});
  renderer.setPixelRatio(Math.min(window.devicePixelRatio,2));
  renderer.setSize(W(), H());
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  mount.appendChild(renderer.domElement);

  const hemi = new THREE.HemisphereLight(0xbfd4ff, 0x1a1f28, 0.7);
  scene.add(hemi);
  const sun = new THREE.DirectionalLight(0xffffff, 1.05);
  sun.position.set(opts.sun[0], opts.sun[1], opts.sun[2]);
  sun.castShadow = true;
  sun.shadow.mapSize.set(1024,1024);
  const fs = opts.fogFar*0.6;
  sun.shadow.camera.left=-fs; sun.shadow.camera.right=fs; sun.shadow.camera.top=fs; sun.shadow.camera.bottom=-fs;
  sun.shadow.camera.near=10; sun.shadow.camera.far=opts.fogFar*2;
  sun.shadow.bias=-0.0018;
  scene.add(sun);

  const root = new THREE.Group();
  scene.add(root);

  const baseGeo = new THREE.CircleGeometry(opts.baseRadius, 64);
  const baseMat = new THREE.MeshStandardMaterial({color:0x141a24, roughness:1});
  const base = new THREE.Mesh(baseGeo, baseMat);
  base.rotation.x = -Math.PI/2;
  base.position.y = -2;
  base.receiveShadow = true;
  root.add(base);

  const grid = new THREE.PolarGridHelper(opts.baseRadius, 8, 5, 48, 0x2a3340, 0x1c2330);
  grid.position.y = -1.8;
  root.add(grid);

  const HEIGHT = opts.height;
  const raycastTargets = [];
  const registry = {};

  regions.forEach(region => {
    const baseColor = new THREE.Color(region.color);
    const topMat = new THREE.MeshStandardMaterial({color: baseColor, roughness:0.75, metalness:0.05, side: THREE.DoubleSide});
    const sideColor = baseColor.clone().multiplyScalar(0.72);
    const sideMat = new THREE.MeshStandardMaterial({color: sideColor, roughness:0.9, metalness:0.02, side: THREE.DoubleSide});

    const entry = { meshes: [], topMat, sideMat, baseColor, baseDepth: HEIGHT, sprite: null };
    registry[region.name] = entry;

    region.polys.forEach(ring => {
      if(ring.length < 3) return;
      const shape = new THREE.Shape();
      shape.moveTo(ring[0][0], ring[0][1]);
      for(let i=1;i<ring.length;i++) shape.lineTo(ring[i][0], ring[i][1]);
      shape.closePath();
      const geo = new THREE.ExtrudeGeometry(shape, {depth:HEIGHT, bevelEnabled:true, bevelThickness:opts.bevel, bevelSize:opts.bevel, bevelSegments:2, steps:1});
      geo.rotateX(-Math.PI/2);
      const mesh = new THREE.Mesh(geo, [topMat, sideMat]);
      mesh.castShadow = true; mesh.receiveShadow = true;
      mesh.userData.name = region.name;
      mesh.userData.type = region.type;
      mesh.userData.topMat = topMat;
      mesh.userData.baseColor = baseColor;
      root.add(mesh);
      raycastTargets.push(mesh);
      entry.meshes.push(mesh);
    });

    if(opts.labels){
      const sprite = makeLabelSprite(region.name, opts.fontSize);
      sprite.position.set(region.label[0], HEIGHT+10, -region.label[1]);
      sprite.renderOrder = 2;
      root.add(sprite);
      entry.sprite = sprite;
    }
  });

  PANEL_REGISTRY[mountId] = registry;

  let radius = opts.radius, theta = opts.theta, phi = opts.phi;
  const target = new THREE.Vector3(0,0,0);
  function updateCam(){
    camera.position.x = target.x + radius*Math.sin(phi)*Math.sin(theta);
    camera.position.y = target.y + radius*Math.cos(phi);
    camera.position.z = target.z + radius*Math.sin(phi)*Math.cos(theta);
    camera.lookAt(target);
  }
  updateCam();

  let dragging=false, lastX=0, lastY=0, autoRotate=false;
  const dom = renderer.domElement;
  dom.addEventListener('pointerdown', e=>{dragging=true;lastX=e.clientX;lastY=e.clientY;autoRotate=false;});
  window.addEventListener('pointermove', e=>{
    if(!dragging) return;
    const dx=e.clientX-lastX, dy=e.clientY-lastY;
    lastX=e.clientX; lastY=e.clientY;
    theta -= dx*0.006; phi -= dy*0.005;
    phi = Math.max(0.15, Math.min(1.35, phi));
    updateCam();
  });
  window.addEventListener('pointerup', ()=>dragging=false);
  dom.addEventListener('wheel', e=>{
    e.preventDefault();
    radius += e.deltaY*(opts.radius*0.0006);
    radius = Math.max(opts.radius*0.35, Math.min(opts.radius*2.2, radius));
    updateCam();
  }, {passive:false});

  const raycaster = new THREE.Raycaster();
  const mouseNDC = new THREE.Vector2();
  let hovered=null, selected=null;
  function setHi(mesh,on){
    if(!mesh) return;
    const c = mesh.userData.baseColor.clone();
    if(on) c.multiplyScalar(1.35);
    mesh.userData.topMat.color.copy(c);
  }
  function pick(cx,cy){
    const rect = dom.getBoundingClientRect();
    mouseNDC.x = ((cx-rect.left)/rect.width)*2-1;
    mouseNDC.y = -((cy-rect.top)/rect.height)*2+1;
    raycaster.setFromCamera(mouseNDC, camera);
    const hits = raycaster.intersectObjects(raycastTargets);
    return hits.length ? hits[0].object : null;
  }
  dom.addEventListener('pointermove', e=>{
    if(dragging){ tooltip.style.display='none'; return; }
    const obj = pick(e.clientX, e.clientY);
    if(obj !== hovered){
      if(hovered && hovered!==selected) setHi(hovered,false);
      hovered = obj;
      if(hovered && hovered!==selected) setHi(hovered,true);
    }
    if(obj){
      tooltip.style.display='block';
      tooltip.style.left = e.clientX+'px';
      tooltip.style.top = e.clientY+'px';
      const t = obj.userData.type==='daegu' ? '대구광역시' : (obj.userData.type==='si' ? '경북 · 시' : '경북 · 군');
      const w = regionWeather[obj.userData.name];
      let html = obj.userData.name + '<span class="sub">'+t+'</span>';
      if(w){
        html += `<span class="weather">체감 ${w.feels}°C · 기온 ${w.temp}°C · 습도 ${w.hum}%<br>${w.stage} (${w.time} 기준)</span>`;
      }
      tooltip.innerHTML = html;
    } else tooltip.style.display='none';
  });
  dom.addEventListener('pointerleave', ()=>{ tooltip.style.display='none'; });
  dom.addEventListener('click', e=>{
    const obj = pick(e.clientX, e.clientY);
    if(selected && selected!==obj) setHi(selected,false);
    selected = (obj===selected)?null:obj;
    if(selected) setHi(selected,true);
  });

  function onResize(){
    camera.aspect = W()/H();
    camera.updateProjectionMatrix();
    renderer.setSize(W(), H());
  }
  window.addEventListener('resize', onResize);

  panels.push({scene, camera, renderer, tick:()=>{ if(autoRotate){ theta+=opts.spin; updateCam(); } }});
}

function heatColorHex(t){
  const stops = [
    [20, [0x3a,0xa0,0xa0]],
    [28, [0xf2,0xd2,0x4b]],
    [31, [0xf7,0xb7,0x33]],
    [33, [0xf2,0x71,0x1c]],
    [35, [0xe0,0x39,0x2b]],
    [38, [0x8b,0x1e,0x1e]],
  ];
  if(t<=stops[0][0]) return rgbHex(stops[0][1]);
  for(let i=0;i<stops.length-1;i++){
    const t0=stops[i][0], c0=stops[i][1], t1=stops[i+1][0], c1=stops[i+1][1];
    if(t>=t0 && t<=t1){
      const f=(t-t0)/(t1-t0);
      const c=c0.map((v,idx)=>Math.round(v+(c1[idx]-v)*f));
      return rgbHex(c);
    }
  }
  return rgbHex(stops[stops.length-1][1]);
}
function rgbHex(c){ return '#'+c.map(v=>v.toString(16).padStart(2,'0')).join(''); }
function heatHeight(t){
  const h = 10 + (t-20)*4.2;
  return Math.max(8, Math.min(95, h));
}

function updateRegionVisual(name, info){
  Object.values(PANEL_REGISTRY).forEach(reg=>{
    const entry = reg[name];
    if(!entry) return;
    const col = new THREE.Color(heatColorHex(info.feels));
    entry.baseColor.copy(col);
    entry.topMat.color.copy(col);
    entry.sideMat.color.copy(col.clone().multiplyScalar(0.7));
    const h = heatHeight(info.feels);
    entry.meshes.forEach(m => { m.scale.y = h/entry.baseDepth; });
    if(entry.sprite) entry.sprite.position.y = h + 10;
  });
}

  [DATA_DAEGU, DATA_GB, DATA_ULLEUNG_MINI].forEach(dataset=>{
    dataset.forEach(region=>{
      Object.values(PANEL_REGISTRY).forEach(reg=>{
        const entry = reg[region.name];
        if(!entry) return;
        const col = new THREE.Color(region.color);
        entry.baseColor.copy(col);
        entry.topMat.color.copy(col);
        entry.sideMat.color.copy(col.clone().multiplyScalar(0.72));
        entry.meshes.forEach(m => { m.scale.y = 1; });
        if(entry.sprite) entry.sprite.position.y = entry.baseDepth + 10;
      });
    });
  });
  Object.keys(regionWeather).forEach(k => delete regionWeather[k]);
  document.getElementById('legend-heat').style.display = 'none';
  document.getElementById('legend-si').style.display = 'flex';
  document.getElementById('legend-gun').style.display = 'flex';
  document.getElementById('upload-status').textContent = '';
  document.getElementById('upload-status').className = '';
  document.getElementById('file-input').value = '';
}

function decodeBuffer(buffer){
  let text = new TextDecoder('utf-8').decode(buffer);
  const bad = (text.match(/\uFFFD/g)||[]).length;
  if(bad > 3){
    try { text = new TextDecoder('euc-kr').decode(buffer); } catch(e){}
  }
  return text;
}

function parseDelimited(text){
  const lines = text.split(/\r?\n/).filter(l=>l.trim().length>0);
  if(!lines.length) return [];
  const delim = lines[0].indexOf('\t')>=0 ? '\t' : ',';
  const headers = lines[0].split(delim).map(h=>h.trim());
  const rows = [];
  for(let i=1;i<lines.length;i++){
    const cols = lines[i].split(delim);
    const obj = {};
    headers.forEach((h,idx)=>{ obj[h] = (cols[idx]!==undefined ? cols[idx] : '').trim(); });
    rows.push(obj);
  }
  return rows;
}

function findKey(headers, candidates){
  for(const c of candidates){
    const hit = headers.find(h => h.indexOf(c) >= 0);
    if(hit) return hit;
  }
  return null;
}

function applyCSVRows(rows){
  if(!rows.length) return {count:0};
  const headers = Object.keys(rows[0]);
  const kRegionGroup = findKey(headers, ['권역']);
  const kLoc = findKey(headers, ['지역']);
  const kFeels = findKey(headers, ['체감온도']);
  const kTemp = findKey(headers, ['기온']);
  const kHum = findKey(headers, ['습도']);
  const kStage = findKey(headers, ['폭염단계']);
  const kTime = findKey(headers, ['예보시간','관측시간','시간']);

  if(!kLoc || !kFeels){
    return {count:0, error:'지역/체감온도 컬럼을 찾을 수 없어요'};
  }

  const filtered = kRegionGroup ? rows.filter(r => (r[kRegionGroup]||'').indexOf('대구')>=0 || (r[kRegionGroup]||'').indexOf('경북')>=0) : rows;

  const peak = {};
  filtered.forEach(r=>{
    const loc = r[kLoc];
    const t = parseFloat(r[kFeels]);
    if(!loc || isNaN(t)) return;
    if(!peak[loc] || t > peak[loc].feels){
      peak[loc] = {
        feels: t,
        temp: kTemp ? parseFloat(r[kTemp]) : null,
        hum: kHum ? parseFloat(r[kHum]) : null,
        stage: kStage ? r[kStage] : '',
        time: kTime ? r[kTime] : ''
      };
    }
  });

  let matched = 0, maxLoc='', maxT=-999;
  Object.keys(peak).forEach(loc=>{
    const targets = CSV_NAME_MAP[loc];
    if(!targets) return;
    const info = peak[loc];
    if(info.feels > maxT){ maxT = info.feels; maxLoc = loc; }
    targets.forEach(name=>{
      regionWeather[name] = info;
      updateRegionVisual(name, info);
    });
    matched++;
  });

  return {count:matched, maxLoc, maxT};
}

document.getElementById('file-input').addEventListener('change', function(e){
  const file = e.target.files[0];
  if(!file) return;
  const statusEl = document.getElementById('upload-status');
  statusEl.className = '';
  statusEl.textContent = '읽는 중...';
  const reader = new FileReader();
  reader.onload = function(evt){
    try{
      const text = decodeBuffer(evt.target.result);
      const rows = parseDelimited(text);
      const result = applyCSVRows(rows);
      if(!result.count){
        statusEl.className = 'err';
        statusEl.textContent = result.error || '반영할 대구·경북 지점을 찾지 못했어요';
        return;
      }
      document.getElementById('legend-si').style.display = 'none';
      document.getElementById('legend-gun').style.display = 'none';
      document.getElementById('legend-heat').style.display = 'flex';
      statusEl.className = '';
      statusEl.textContent = `${result.count}개 지점 반영됨 · 최고 체감 ${result.maxT}°C (${result.maxLoc})`;
    } catch(err){
      statusEl.className = 'err';
      statusEl.textContent = '파일을 읽는 중 오류가 발생했어요';
      console.error(err);
    }
  };
  reader.readAsArrayBuffer(file);
});
