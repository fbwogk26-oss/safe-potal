const HEIGHT = 16;
const raycastTargets = [];
const regionGroups = {};

REGIONS.forEach(region => {
  const grp = new THREE.Group();
  grp.userData.name = region.name;
  grp.userData.type = region.type;

  const baseColor = new THREE.Color(region.color);
  const topMat = new THREE.MeshStandardMaterial({color: baseColor, roughness:0.75, metalness:0.05, side: THREE.DoubleSide});
  const sideColor = baseColor.clone().multiplyScalar(0.72);
  const sideMat = new THREE.MeshStandardMaterial({color: sideColor, roughness:0.9, metalness:0.02, side: THREE.DoubleSide});

  region.polys.forEach(ring => {
    if(ring.length < 3) return;
    const shape = new THREE.Shape();
    shape.moveTo(ring[0][0], ring[0][1]);
    for(let i=1;i<ring.length;i++) shape.lineTo(ring[i][0], ring[i][1]);
    shape.closePath();

    const geo = new THREE.ExtrudeGeometry(shape, {
      depth: HEIGHT, bevelEnabled: true, bevelThickness: 1.4, bevelSize: 1.2, bevelSegments: 2, steps:1
    });
    geo.rotateX(-Math.PI/2);

    const mats = [topMat, sideMat];
    const mesh = new THREE.Mesh(geo, mats);
    mesh.castShadow = true;
    mesh.receiveShadow = true;
    mesh.userData.name = region.name;
    mesh.userData.type = region.type;
    mesh.userData.topMat = topMat;
    mesh.userData.baseColor = baseColor.clone();
    grp.add(mesh);
    raycastTargets.push(mesh);
  });

  root.add(grp);
  regionGroups[region.name] = grp;
});

function makeLabelSprite(text, small){
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  const fontSize = small ? 30 : 34;
  ctx.font = `700 ${fontSize}px 'Malgun Gothic', sans-serif`;
  const padX = 22;
  const w = Math.ceil(ctx.measureText(text).width) + padX*2;
  const h = fontSize + 26;
  canvas.width = w*2; canvas.height = h*2;
  ctx.scale(2,2);
  ctx.font = `700 ${fontSize}px 'Malgun Gothic', sans-serif`;
  ctx.textBaseline = 'middle';
  ctx.textAlign = 'center';

  ctx.fillStyle = 'rgba(15,18,24,0.72)';
  const r = 10;
  ctx.beginPath();
  ctx.moveTo(r,0);
  ctx.arcTo(w,0,w,h,r);
  ctx.arcTo(w,h,0,h,r);
  ctx.arcTo(0,h,0,0,r);
  ctx.arcTo(0,0,w,0,r);
  ctx.closePath();
  ctx.fill();

  ctx.fillStyle = '#f2f5f8';
  ctx.fillText(text, w/2, h/2+1);

  const tex = new THREE.CanvasTexture(canvas);
  tex.minFilter = THREE.LinearFilter;
  const mat = new THREE.SpriteMaterial({map:tex, depthTest:true, depthWrite:false, transparent:true});
  const sprite = new THREE.Sprite(mat);
  const scale = 0.30;
  sprite.scale.set(w*scale, h*scale, 1);
  return sprite;
}

});

let radius = 1450, theta = Math.PI*0.28, phi = Math.PI*0.34;
const target = new THREE.Vector3(0,0,0);

function updateCamera(){
  camera.position.x = target.x + radius*Math.sin(phi)*Math.sin(theta);
  camera.position.y = target.y + radius*Math.cos(phi);
  camera.position.z = target.z + radius*Math.sin(phi)*Math.cos(theta);
  camera.lookAt(target);
}
updateCamera();

let dragging = false, lastX=0, lastY=0;
let autoRotate = true;

function onDown(x,y){ dragging=true; lastX=x; lastY=y; autoRotate=false; }
function onMove(x,y){
  if(!dragging) return;
  const dx = x-lastX, dy = y-lastY;
  lastX=x; lastY=y;
  theta -= dx*0.0055;
  phi   -= dy*0.0045;
  phi = Math.max(0.12, Math.min(1.35, phi));
  updateCamera();
}
function onUp(){ dragging=false; }

renderer.domElement.addEventListener('pointerdown', e=>onDown(e.clientX,e.clientY));
window.addEventListener('pointermove', e=>onMove(e.clientX,e.clientY));
window.addEventListener('pointerup', onUp);
renderer.domElement.addEventListener('wheel', e=>{
  e.preventDefault();
  radius += e.deltaY*0.6;
  radius = Math.max(420, Math.min(2600, radius));
  updateCamera();
}, {passive:false});

const raycaster = new THREE.Raycaster();
const mouseNDC = new THREE.Vector2();
const tooltip = document.getElementById('tooltip');
let hovered = null;
let selected = null;

function setHighlight(mesh, on){
  if(!mesh) return;
  const c = mesh.userData.baseColor.clone();
  if(on) c.multiplyScalar(1.35);
  mesh.userData.topMat.color.copy(c);
}

function pick(clientX, clientY){
  mouseNDC.x = (clientX/window.innerWidth)*2-1;
  mouseNDC.y = -(clientY/window.innerHeight)*2+1;
  raycaster.setFromCamera(mouseNDC, camera);
  const hits = raycaster.intersectObjects(raycastTargets);
  return hits.length ? hits[0].object : null;
}

renderer.domElement.addEventListener('pointermove', e=>{
  if(dragging){ tooltip.style.display='none'; return; }
  const obj = pick(e.clientX, e.clientY);
  if(obj !== hovered){
    if(hovered && hovered!==selected) setHighlight(hovered,false);
    hovered = obj;
    if(hovered && hovered!==selected) setHighlight(hovered,true);
  }
  if(obj){
    tooltip.style.display='block';
    tooltip.style.left = e.clientX+'px';
    tooltip.style.top = e.clientY+'px';
    const label = obj.userData.type==='daegu' ? '대구광역시' : (obj.userData.type==='si' ? '경북 · 시' : '경북 · 군');
    tooltip.innerHTML = obj.userData.name + '<span>'+label+'</span>';
  } else {
    tooltip.style.display='none';
  }
});

renderer.domElement.addEventListener('click', e=>{
  const obj = pick(e.clientX, e.clientY);
  if(selected && selected!==obj) setHighlight(selected,false);
  selected = (obj===selected) ? null : obj;
  if(selected) setHighlight(selected,true);
});

window.addEventListener('resize', ()=>{
  camera.aspect = window.innerWidth/window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});

const needle = document.getElementById('compass-needle');

}

  requestAnimationFrame(animate);
  if(autoRotate){
    theta += 0.0016;
    updateCamera();
  }
  updateCompass();
  renderer.render(scene, camera);
}
